\section{Release v2.0.0 (2026-11-25)}

\subsection{Key New Features}
\begin{enumerate}
    \item \textbf{Normative Spec Upgrade --- \texttt{IFOG-SIM-SPEC-005} (v5.0) \& 41-Formula Verification:}
    Added checksum-locked \texttt{spec/IFOG-SIM-SPEC-005.\{tex,pdf,sha256\}} (retaining \texttt{IFOG-SIM-SPEC-004} byte-identical), backward-compatible \texttt{configs/schema\_v5.json}, and \texttt{configs/baseline\_1km\_6state.json}. Expanded per-formula CI traceability to \textbf{41 / 41 independent tests} (\texttt{EQ-01}--\texttt{EQ-41}).
    \item \textbf{Official 6-State Symmetric DSP (\texttt{EQ-31}--\texttt{EQ-34}, Gate 5):}
    Implemented anti-periodic 6-state symmetric modulation ($\alpha^* = \pi/3$) in \texttt{DigitalDSP\_SC} with 3-channel orthogonal demodulation ($e_\Omega^{(6)}, e_{V_\pi}^{(6)}, e_{\text{asym}}^{(6)}$) and Loop-1 anti-windup clamping. Verified $-33.3\%$ peak MIOC voltage ($0.667\,V_\pi$), static error $1.28\times 10^{-16}\text{ rad}$, $\epsilon_{V_\pi,\text{eff}}=1.31\times 10^{-11}$, and \textbf{$+136.94\text{ dB}$ ($7.03\times 10^6\times$) RAM/crosstalk/$\gamma_{\text{nl}}$ rejection} ($354.56 \to 5.04\times 10^{-5}\,^\circ/\text{h}$).
    \item \textbf{Group A --- 6 Official Time-Domain Scenarios (\texttt{TD-01}--\texttt{TD-06}, \texttt{--mode td\_all}):}
    Unlocked 64 B \texttt{dsp\_trace.bin} and \texttt{micro\_waveforms.bin} across \texttt{td\_blanking} ($t_s=174.1\text{ ns}$, $k^*_{\text{start}}=5$), \texttt{td\_step\_sine} ($t_r=0.22\text{ ms}$, $-3\text{ dB}$ BW $=2{,}540\text{ Hz}$ vs theory $2{,}354\text{ Hz}$), \texttt{td\_vpi\_shock} ($+5\%$ $V_\pi$ shock, $2\times$ lower Loop-1 transient in 6-state), \texttt{td\_high\_rate} ($700^\circ/\text{s}$ RMS err $0.00028\%$, $7{,}184$ wraps @ $179.6\text{ kHz}$ + anti-windup), \texttt{td\_shupe\_shock} ($1{,}483.6\,^\circ/\text{h}$ shock, $99.74\%$ filter suppression), and \texttt{td\_limit\_cycles} (12-bit PRBS dither unlocking $3\to 1{,}595$ unique DAC codes).
    \item \textbf{Group B --- 5 Official System Engineering \& Production Scenarios (\texttt{ENG-01}--\texttt{ENG-05}, \texttt{--mode eng\_all}):}
    Added \texttt{eng\_hw\_sizing} (min ENOB $=12$-bit, aperture jitter \texttt{EQ-35}, TIA/SLD Pareto ARW $1.412\times 10^{-4}\,^\circ/\sqrt{\text{h}}$), \texttt{eng\_imu\_3axis} (real concurrent 3-axis SystemC-AMS IMU \texttt{--imu3axis}, rail droop $92.4\text{ mV}$, Axis-Y crosstalk $40.63 \to 0.106\,^\circ/\text{h} = +51.6\text{ dB}$ rejection), \texttt{eng\_vre\_faraday} (pure $G_2$ VRE $R^2=0.9998$, Faraday $0.183\,^\circ/\text{h/G}$, Mu-metal $25.25\text{ dB}$), \texttt{eng\_fault\_bit} ($5/5$ faults detected $\le 30.0\ \mu\text{s}$ + SEU FDIR soft-recovery), and \texttt{eng\_mc\_yield\_cal} ($N=500$ MC Yield $=97.2\%$, $C_{pk}=0.604$; $-40..+85^\circ\text{C}$ CSV + Poly $99.98\%$ / 4-state EKF $99.95\%$).
    \item \textbf{Unified Batch/CLI Flow, Official Scenarios Manual \& Governance Sync (Rules 6 \& 16):}
    Added \texttt{scripts/run\_all\_scenarios.sh}, 22 CLI modes in \texttt{sim\_runner\_cli.py}, 300-DPI plotters (\texttt{viz/plot\_scenarios.py}), \texttt{docs/IFOG\_SIMULATION\_SCENARIOS.\{md,tex,pdf\}}, and skill \texttt{skills/ifog-6state-and-scenarios/SKILL.md} (8 skills total).
\end{enumerate}
