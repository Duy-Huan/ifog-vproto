\section{Release v4.1.0 (2026-11-27)}

\subsection{Key New Features (Group E TRL 7--8 / DO-254 Industrial Suite, 2/4/6-State FPGA DSP \& Multi-Instance FMU)}
\begin{enumerate}
    \item \textbf{\texttt{IND-01} Physical Lab Hardware Data Ingestion \& NASA-STD-7009A CAS-4 / ASME V\&V 20 (\texttt{lab\_validation.py}):}
    Standardized 3-rig CSV ingestor (\texttt{configs/lab\_datasets/reference\_ifog\_lab/}: 10 h granite-table AVAR, $[-2,+2]^\circ/\text{s}$ slow rate-table Deadband/Bunny Ears, $-40..+85^\circ\text{C}$ thermal chamber) + CLI \texttt{--lab-dir}; cuts uncalibrated datasheet error ($101.9\%$) to \textbf{$4.51\%$} ($<5\%$), achieving ASME V\&V 20 max $|E|/u_{\text{val}} = 0.723 \le 1.0$ (\textbf{CAS-4}) and Tier 2 C++ tracking error $0.0031\%$.
    \item \textbf{\texttt{IND-02} Bit-True FPGA Fixed-Point DSP (\texttt{Q16.32}) \& Synthesizable SystemVerilog RTL (\texttt{rtl/ifog\_dsp\_6state.sv}):}
    Exact signed $Q_{(W-F).F}$ saturating/truncating register quantization in \texttt{DigitalDSP\_SC} and SV-2012 RTL supporting selectable \textbf{2-State, 4-State, and 6-State} modes; achieves \textbf{0-LSB} RTL match, 0 overflows, and $31\times$--$376\times$ lower quantization noise than 32-bit \texttt{Q8.24}.
    \item \textbf{\texttt{IND-03} 3D Cylindrical ANSYS/COMSOL FEM $\to$ \texttt{shupe\_lut.bin} v2 (\texttt{shupe\_thermal.py}):}
    Smooth linear RBF 3D cylindrical projection of eccentric housing hotspots $T(r,\theta,z,t)$ \& anisotropic potting stress $\sigma_{\text{eff}}(r,\theta,z,t)$ onto discrete fiber turns (\texttt{--fem-3d-csv}), proving \textbf{$6.00\times$} Octupole vs Quadrupole 3D suppression in real Tier 2 C++.
    \item \textbf{\texttt{IND-04} MIL-STD-810H / DO-160G Broadband Random Vibration PSD \& 2-DOF Resonance:}
    Piecewise log-log PSD ($20$--$2000\text{ Hz}$, $4.88\text{ g}_{\text{rms}}$) through 2-DOF elastomer isolator ($65\text{ Hz}$) + bobbin ($480\text{ Hz}, Q=12$) transmissibility + Miles' Eq cross-check, achieving \textbf{$4.82\times$} coil $g_{\text{rms}}$ attenuation and \textbf{$3.11\times$} Tier 2 C++ VRE bias reduction.
    \item \textbf{\texttt{IND-05} Multi-Instance Thread-Safe Pure-C++17 Source-Code FMI 3.0.2 FMU (\texttt{build/ifog\_core\_multi.fmu}):}
    Eliminates Accellera SystemC singleton lock (\texttt{canBeInstantiatedOnlyOncePerProcess="false"}), bundles portable \texttt{sources/buildDescription.xml} for Windows/Simulink \& RT-Linux, passes FMPy validation (0 XSD errors), and executes \textbf{3 concurrent axes (6-State / 4-State / 2-State) in 1 OS process} ($<0.15\%$ error, 0 cross-talk).
    \item \textbf{\texttt{IND-06} 12-Corner Worst-Case PVT Matrix \& 10-Yr SLD/MIOC Aging + Space 50 krad(Si) TID:}
    Evaluates $-40/+25/+85^\circ\text{C} \times \pm 5\%$ rail $\times \pm 3\sigma$ process + exponential SLD droop \& PM fiber RIA(TID) via real Tier 1 + Tier 2 C++ milestones (WC PVT ARW $0.00049$, EOL ARW $0.00052\text{ deg}/\sqrt{\text{h}}$).
    \item \textbf{Complete 2/4/6-State DSP Coverage, 29/29 Suite, 100\% Dual Coverage \& 23 Release Assets:}
    Extended \texttt{DSP-01}, \texttt{DYN-01..03}, \texttt{IND-02}, and \texttt{IND-05} across all three DSP modes (2/4/6-State); verified \textbf{100.0\% C++} ($2{,}733/2{,}733$ lines, $299/299$ functions) \& \textbf{100.00\% Python} ($5{,}632/5{,}632$ stmts) coverage, \textbf{29/29 PASS} curated example, synchronized 38-slide presentation deck, and \textbf{23 verified release assets}.
\end{enumerate}
